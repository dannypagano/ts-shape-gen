import React, { useState, useCallback, useMemo, useRef } from "react";
import Sketch from "./Sketch.jsx";

const Field = ({ label, children }) => (
  <div className="space-y-1">
    <label className="block text-xs font-medium text-zinc-300">{label}</label>
    {children}
  </div>
);

const Num = (props) => (
  <input
    type="number"
    {...props}
    className="hs-input w-full rounded-md bg-zinc-900/60 border border-zinc-700/60 px-2 py-1 text-sm outline-none focus:border-emerald-500 focus:ring-1 focus:ring-emerald-500"
  />
);

const Toggle = ({ checked, onChange }) => (
  <input
    type="checkbox"
    className="hs-checkbox rounded border-zinc-700/60"
    checked={checked}
    onChange={onChange}
  />
);

function PaletteRow({ idx, value, onChange, onRemove, lock, isLocked }) {
  const [bg, a, b] = value;
  return (
    <div className="flex items-center gap-2 p-2 rounded-lg border border-zinc-800/80 bg-zinc-900/40">
      <div className="flex gap-1">
        <input
          type="color"
          value={bg}
          onChange={(e) => onChange(idx, [e.target.value, a, b])}
          title="Background"
          className="h-8 w-8 rounded"
        />
        <input
          type="color"
          value={a}
          onChange={(e) => onChange(idx, [bg, e.target.value, b])}
          title="Shape 1"
          className="h-8 w-8 rounded"
        />
        <input
          type="color"
          value={b}
          onChange={(e) => onChange(idx, [bg, a, e.target.value])}
          title="Shape 2"
          className="h-8 w-8 rounded"
        />
      </div>

      <div className="ml-auto flex items-center gap-2">
        <button
          type="button"
          onClick={() => lock(idx)}
          title="Lock to this palette"
          className={`hs-btn px-2 py-1 text-xs rounded-md border ${
            isLocked
              ? "bg-emerald-600 border-emerald-500 text-white"
              : "bg-zinc-800 border-zinc-700 text-zinc-200 hover:bg-zinc-700"
          }`}
        >
          {isLocked ? "Locked" : "Lock"}
        </button>
        <button
          type="button"
          onClick={() => onRemove(idx)}
          className="hs-btn px-2 py-1 text-xs rounded-md border bg-zinc-800 border-zinc-700 text-zinc-200 hover:bg-zinc-700"
          title="Remove palette"
        >
          Remove
        </button>
      </div>
    </div>
  );
}

export default function App() {
  // Canvas + grid
  const [width, setWidth] = useState(700);
  const [height, setHeight] = useState(700);
  const [targetCell, setTargetCell] = useState(70);
  const [showGrid, setShowGrid] = useState(false);

  // Two-cell limits
  const [maxPills, setMaxPills] = useState(5);
  const [maxHalves, setMaxHalves] = useState(3);

  // Weights
  const [wPill, setWPill] = useState(2);
  const [wHalf, setWHalf] = useState(2);
  const [wSquare, setWSquare] = useState(2);
  const [wPie, setWPie] = useState(2);
  const [wCircle, setWCircle] = useState(2);
  const [wBlank, setWBlank] = useState(5);

  // Palettes (editable)
  const [palettes, setPalettes] = useState([
    ["#0B1021", "#F8E16C", "#FF6B6B"],
    ["#111111", "#7FDBDA", "#FF6B6B"],
    ["#FAF3DD", "#355070", "#E56B6F"],
    ["#0D1B2A", "#E0E1DD", "#778DA9"],
    ["#F7F7F2", "#333533", "#F5CB5C"],
  ]);
  const [lockPalette, setLockPalette] = useState(false);
  const [lockedIndex, setLockedIndex] = useState(0);

  const addPalette = () => {
    setPalettes((p) => [...p, ["#0B1021", "#F8E16C", "#FF6B6B"]]);
    setLockedIndex((i) => (lockPalette ? palettes.length : i));
  };
  const removePalette = (idx) => {
    setPalettes((p) => p.filter((_, i) => i !== idx));
    if (lockedIndex === idx) setLockedIndex(0);
    if (palettes.length <= 1) setLockPalette(false);
  };
  const updatePalette = (idx, value) => {
    setPalettes((p) => p.map((row, i) => (i === idx ? value : row)));
  };
  const lockTo = (idx) => {
    setLockedIndex(idx);
    setLockPalette(true);
  };

  const [params, setParams] = useState({
    width: 700,
    height: 700,
    targetCell: 70,
    showGrid: false,
    maxPills: 5,
    maxHalves: 3,
    weights: { pill: 2, half: 2, square: 2, pie: 2, circle: 2, blank: 5 },
    palettes: [
      ["#19224A", "#3F5DB3", "#6C94EC"], // Blue
      ["#082429", "#0D4B3B", "#1EA672"], // Green
      ["#3A1607", "#762B0B", "#D97917"], // Orange
      ["#420000", "#940821", "#E46C63"], // Red
      ["#27103C", "#673C87", "#AE74D8"], // Purple
    ],
    lockPalette: false,
    lockedIndex: 0,
    randomizePalette: true, // NEW
  });
  [
    width,
    height,
    targetCell,
    showGrid,
    maxPills,
    maxHalves,
    wPill,
    wHalf,
    wSquare,
    wPie,
    wCircle,
    wBlank,
    palettes,
    lockPalette,
    lockedIndex,
  ];

  const resetTo700 = useCallback(() => {
    setWidth(700);
    setHeight(700);
    setTargetCell(70);
  }, []);

  const sketchRef = useRef(null);
  const handleGenerate = () => {
    sketchRef.current?.generate(params);
  };
  // Build the CURRENT params from the individual control states
  const finalParams = {
    width,
    height,
    targetCell,
    showGrid,
    maxPills,
    maxHalves,
    weights: {
      pill: wPill,
      half: wHalf,
      maxPills,
      maxHalves,
      weights: {
        pill: wPill,
        half: wHalf,
        square: wSquare,
        pie: wPie,
        circle: wCircle,
        blank: wBlank,
      },
    },

    // palettes come from your editable/selector state.
    // If you want to keep them editable in the future, use that source here.
    palettes: params.palettes, // or a dedicated `palettes` state if you re-enable editor
    lockPalette,
    lockedIndex,
    randomizePalette: params.randomizePalette ?? true,
  };
  sketchRef.current?.generate(finalParams);
}

return (
  <div className="min-h-screen grid grid-cols-1 lg:grid-cols-[380px_1fr] bg-zinc-950 text-zinc-100">
    {/* Sidebar */}
    <aside className="border-b lg:border-b-0 lg:border-r border-zinc-800 bg-zinc-900/40 p-5 space-y-6">
      <div className="flex items-start justify-between">
        <div>
          <h1 className="text-xl font-semibold">Grid Shapes Generator</h1>
          <p className="text-xs text-zinc-400 mt-1">
            Preline + Tailwind UI • p5.js canvas
          </p>
        </div>
        <button
          type="button"
          onClick={handleGenerate}
          className="hs-btn px-3 py-2 rounded-md bg-emerald-600 hover:bg-emerald-500"
          title="Generate a new layout"
        >
          Generate
        </button>
      </div>

      <div className="grid gap-4">
        <div>
          {/* Canvas & Grid */}
          <div id="acc-basic">
            <h3 className="text-md font-semibold text-zinc-100">
              Canvas & Grid
            </h3>
            <div>
              <div className="pt-3 grid grid-cols-2 gap-3">
                <Field label="Canvas width">
                  <Num
                    value={width}
                    onChange={(e) => setWidth(+e.target.value || 0)}
                  />
                </Field>
                <Field label="Canvas height">
                  <Num
                    value={height}
                    onChange={(e) => setHeight(+e.target.value || 0)}
                  />
                </Field>
                <Field label="Target cell (px)">
                  <Num
                    value={targetCell}
                    onChange={(e) =>
                      setTargetCell(Math.max(1, +e.target.value || 1))
                    }
                  />
                </Field>
                <Field label="Show grid">
                  <div className="flex items-center gap-2">
                    <Toggle
                      checked={showGrid}
                      onChange={(e) => setShowGrid(e.target.checked)}
                    />
                    <button
                      className="hs-btn text-xs px-2 py-1 rounded-md bg-zinc-800 border border-zinc-700 hover:bg-zinc-700"
                      onClick={resetTo700}
                    >
                      700×700
                    </button>
                  </div>
                </Field>
              </div>
            </div>
          </div>
        </div>

        {/* 2-cell limits */}
        <div className="mt-3" id="acc-limits">
          <h3 className="text-md font-semibold text-zinc-100">
            Pill and Half Circle Limits
          </h3>
          <div>
            <div className="pt-3 grid grid-cols-2 gap-3">
              <Field label="Max no. of pills">
                <Num
                  value={maxPills}
                  onChange={(e) =>
                    setMaxPills(Math.max(0, +e.target.value || 0))
                  }
                />
              </Field>
              <Field label="Max no. of half-circles">
                <Num
                  value={maxHalves}
                  onChange={(e) =>
                    setMaxHalves(Math.max(0, +e.target.value || 0))
                  }
                />
              </Field>
            </div>
          </div>
        </div>

        {/* Weights */}
        <div id="acc-weights">
          <h3 className="text-md font-semibold text-zinc-100">Shape Weights</h3>
          <div>
            <div className="pt-3 grid grid-cols-2 gap-3">
              <Field label="Pill">
                <Num
                  value={wPill}
                  onChange={(e) => setWPill(Math.max(0, +e.target.value || 0))}
                />
              </Field>
              <Field label="Half-circle">
                <Num
                  value={wHalf}
                  onChange={(e) => setWHalf(Math.max(0, +e.target.value || 0))}
                />
              </Field>
              <Field label="Square">
                <Num
                  value={wSquare}
                  onChange={(e) =>
                    setWSquare(Math.max(0, +e.target.value || 0))
                  }
                />
              </Field>
              <Field label="Quarter Circle">
                <Num
                  value={wPie}
                  onChange={(e) => setWPie(Math.max(0, +e.target.value || 0))}
                />
              </Field>
              <Field label="Circle">
                <Num
                  value={wCircle}
                  onChange={(e) =>
                    setWCircle(Math.max(0, +e.target.value || 0))
                  }
                />
              </Field>
              <Field label="Blank">
                <Num
                  value={wBlank}
                  onChange={(e) => setWBlank(Math.max(0, +e.target.value || 0))}
                />
              </Field>
            </div>
          </div>
        </div>
        {/* === Palette Selector === */}
        <div>
          <h3 className="text-md font-semibold text-zinc-100">Palette</h3>

          <div className="mt-3 space-y-2">
            {[
              { name: "Blue", colors: ["#19224A", "#3F5DB3", "#6C94EC"] },
              { name: "Green", colors: ["#082429", "#0D4B3B", "#1EA672"] },
              { name: "Orange", colors: ["#3A1607", "#762B0B", "#D97917"] },
              { name: "Red", colors: ["#420000", "#940821", "#E46C63"] },
              { name: "Purple", colors: ["#27103C", "#673C87", "#AE74D8"] },
            ].map((p, i) => (
              <button
                key={i}
                onClick={() =>
                  setParams((prev) => ({
                    ...prev,
                    lockPalette: true,
                    lockedIndex: i,
                    randomizePalette: false,
                  }))
                }
                className={`w-full flex items-center justify-between px-3 py-2 rounded-lg border transition 
          ${
            params.lockPalette && params.lockedIndex === i
              ? "border-blue-400 bg-blue-900/20"
              : "border-zinc-700 hover:border-zinc-500"
          }`}
              >
                <span className="text-sm text-zinc-200">{p.name}</span>
                <div className="flex space-x-1">
                  {p.colors.map((c, j) => (
                    <div
                      key={j}
                      className="w-4 h-4 rounded"
                      style={{ backgroundColor: c }}
                    />
                  ))}
                </div>
              </button>
            ))}
          </div>

          {/* Randomize Toggle */}
          <div className="flex items-center justify-between mt-4">
            <label className="text-sm text-zinc-300">
              Randomize Palette Each Time
            </label>
            <input
              type="checkbox"
              checked={params.randomizePalette}
              onChange={(e) =>
                setParams((prev) => ({
                  ...prev,
                  randomizePalette: e.target.checked,
                  lockPalette: !e.target.checked,
                }))
              }
              className="h-4 w-4 accent-blue-500"
            />
          </div>
        </div>
      </div>
    </aside>
    {/* End Sidebar */}

    {/* Canvas area */}

    <main className="p-4 lg:p-6">
      <div className="rounded-2xl border border-zinc-800 bg-zinc-900/40 p-3">
        {/* Pass a ref so we can call .generate(params) */}
        <Sketch ref={sketchRef} />
      </div>
    </main>
  </div>
);
